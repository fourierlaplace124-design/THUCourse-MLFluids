\section{计算流体力学基础}
\label{sec:basics}
本讲主线：\textbf{控制方程 $\to$ PDE 分类与信息传播 $\to$ 差分格式与稳定性 $\to$ 计算网格}。

\subsection{控制方程和矢量形式控制方程}
\label{sec:cfd-equations}
本节依次掌握\textbf{四类守恒规律、守恒形式、矢量通量}。
记速度 $\mathbf v=(u,v,w)^{\mathsf T}$，密度 $\rho$，压强 $p$，比内能 $e$，
单位体积总能 $E_t=\rho(e+|\mathbf v|^2/2)$，物质导数
$D/Dt=\partial_t+\mathbf v\cdot\nabla$。

\paragraph{1. 四类控制方程}
以下能量方程按单组分流体写，组分方程列出多组分输运的基本形式：
\begin{align}
 \text{连续：}\quad&\frac{D\rho}{Dt}+\rho\nabla\cdot\mathbf v=0,\\
 \text{动量：}\quad&\rho\frac{D\mathbf v}{Dt}
       =\rho\mathbf g-\nabla p+\nabla\cdot\boldsymbol\tau,\\
 \text{内能：}\quad&\rho\frac{De}{Dt}
       =\rho\dot q-p\nabla\cdot\mathbf v-\nabla\cdot\mathbf q
        +\boldsymbol\tau:\nabla\mathbf v,\\
 \text{组分：}\quad&\rho\frac{DY_s}{Dt}
       =-\nabla\cdot\mathbf j_s+\dot\omega_s.
 \label{eq:cfd-material-laws}
\end{align}
\begin{itemize}
 \item \textbf{连续：}密度变化与体积膨胀平衡；不可压流动满足 $\nabla\cdot\mathbf v=0$。
 \item \textbf{动量：}惯性变化 $=$ 体力 $+$ 压力梯度力 $+$ 黏性力。
 \item \textbf{内能：}内能变化 $=$ 体热源 $+$ 压缩功 $+$ 热通量净输入 $+$ 黏性耗散。
       其中 $\mathbf q=-k\nabla T$，$\boldsymbol\tau:\nabla\mathbf v$ 为双点积。
 \item \textbf{组分：}$Y_s$ 为质量分数，$\mathbf j_s=\rho Y_s\mathbf V_s$ 为扩散通量，
       $\dot\omega_s$ 为反应质量源；$N_s$ 个组分满足 $\sum_sY_s=1$，只有 $N_s-1$ 个独立组分方程。
\end{itemize}
这里 $\mathbf g$ 为单位质量体力，$\dot q$ 为单位质量加热率，$k$ 为导热系数，
$\mathbf V_s$ 为相对质量平均速度的组分扩散速度。

\paragraph{2. 守恒形式}
\[
 \boxed{\text{守恒量的局部积累}+\text{通量净流出}=\text{源项}}
\]
对应的质量、动量、\textbf{总能}和组分方程为
\begin{align}
 &\rho_t+\nabla\cdot(\rho\mathbf v)=0,\\
 &(\rho\mathbf v)_t+\nabla\cdot
    (\rho\mathbf v\otimes\mathbf v+p\mathbf I-\boldsymbol\tau)=\rho\mathbf g,\\
 &(E_t)_t+\nabla\cdot
    [(E_t+p)\mathbf v-\boldsymbol\tau\cdot\mathbf v+\mathbf q]
    =\rho\mathbf g\cdot\mathbf v+\rho\dot q,\\
 &(\rho Y_s)_t+\nabla\cdot(\rho Y_s\mathbf v+\mathbf j_s)=\dot\omega_s.
 \label{eq:cfd-conservative-laws}
\end{align}
$\otimes$ 为张量积。\textbf{非守恒形式强调随体变化，守恒形式强调守恒量与通量。}
光滑解下，利用连续方程有
\[
 (\rho\phi)_t+\nabla\cdot(\rho\mathbf v\phi)=\rho\frac{D\phi}{Dt}.
\]
两种形式可相互转换；含激波时应使用积分／弱意义的守恒律。
内能式与总能式不能混用：\textbf{CFD 的强守恒能量变量通常取 $E_t$}。

\paragraph{3. 矢量守恒形式}
单组分三维流动写为
\begin{equation}
 \boxed{\mathbf U_t+\mathbf F_x+\mathbf G_y+\mathbf H_z=\mathbf S},
 \qquad
 \mathbf U=\begin{bmatrix}\rho&\rho u&\rho v&\rho w&E_t\end{bmatrix}^{\mathsf T}.
 \label{eq:cfd-vector-conservation}
\end{equation}
$\mathbf F,\mathbf G,\mathbf H$ 分别为三个方向的通量；$\mathbf S$ 为体源向量：
\[
 \mathbf S=\begin{bmatrix}
 0&\rho g_x&\rho g_y&\rho g_z&\rho\mathbf g\cdot\mathbf v+\rho\dot q
 \end{bmatrix}^{\mathsf T}.
\]
\textbf{通量 $=$ 无黏部分 $-$ 黏性部分}，即
$\mathbf F=\mathbf F_I-\mathbf F_V$，$\mathbf G=\mathbf G_I-\mathbf G_V$，
$\mathbf H=\mathbf H_I-\mathbf H_V$。
\begin{equation}
 \mathbf F_I=\begin{bmatrix}\rho u\\\rho u^2+p\\\rho uv\\\rho uw\\u(E_t+p)\end{bmatrix},\quad
 \mathbf G_I=\begin{bmatrix}\rho v\\\rho uv\\\rho v^2+p\\\rho vw\\v(E_t+p)\end{bmatrix},\quad
 \mathbf H_I=\begin{bmatrix}\rho w\\\rho uw\\\rho vw\\\rho w^2+p\\w(E_t+p)\end{bmatrix}.
 \label{eq:cfd-inviscid-fluxes}
\end{equation}
\begin{equation}
 \mathbf F_V=\begin{bmatrix}0\\\tau_{xx}\\\tau_{yx}\\\tau_{zx}\\\Psi_x\end{bmatrix},\quad
 \mathbf G_V=\begin{bmatrix}0\\\tau_{xy}\\\tau_{yy}\\\tau_{zy}\\\Psi_y\end{bmatrix},\quad
 \mathbf H_V=\begin{bmatrix}0\\\tau_{xz}\\\tau_{yz}\\\tau_{zz}\\\Psi_z\end{bmatrix},
 \label{eq:cfd-viscous-fluxes}
\end{equation}
其中能量输运分量全部展开为
\begin{align*}
 \Psi_x&=u\tau_{xx}+v\tau_{yx}+w\tau_{zx}-q_x,\\
 \Psi_y&=u\tau_{xy}+v\tau_{yy}+w\tau_{zy}-q_y,\\
 \Psi_z&=u\tau_{xz}+v\tau_{yz}+w\tau_{zz}-q_z.
\end{align*}

黏性应力满足 $\tau_{ij}=\tau_{ji}$。牛顿流体在 Stokes 假设下取（$\mu$ 为动力黏度）
\[
 \boldsymbol\tau=\mu\left[\nabla\mathbf v+(\nabla\mathbf v)^{\mathsf T}
                 -\tfrac23(\nabla\cdot\mathbf v)\mathbf I\right].
\]
\textbf{Euler 方程：}略去黏性与热传导；无体源时另取 $\mathbf S=0$。
量热完全理想气体的闭合关系为
$p=(\gamma-1)[E_t-\rho|\mathbf v|^2/2]$，$\gamma=c_p/c_v$。

\textbf{边界条件：}黏性固壁通常无滑移、不可穿透；Euler 固壁只要求不可穿透。
热边界可给定壁温或热流，绝热对应 $\mathbf q\cdot\mathbf n=0$。

\subsection{偏微分方程的数学性质}
\label{sec:cfd-pde}
\textbf{拟线性：最高阶导数线性出现。}下面用二元一阶系统讨论特征；
Euler 方程是一阶系统，Navier--Stokes 方程还含黏性二阶导数。

\paragraph{1. 特征线与依赖域}
\textbf{特征线}是控制方程的主部发生退化、数据须满足相容关系的方向轨迹。
对 $u_t+cu_x=0$，沿 $\dd x/\dd t=c$ 有 $\dd u/\dd t=0$，即信息沿特征线传播。
\textbf{依赖域}决定某点的解值，\textbf{影响域}表示该点扰动能够到达的区域。

\paragraph{2. 判别式法}
对方程组
\begin{equation}
 \begin{aligned}
 a_1u_x+b_1u_y+c_1v_x+d_1v_y&=f_1,\\
 a_2u_x+b_2u_y+c_2v_x+d_2v_y&=f_2,
 \end{aligned}
 \label{CFD:LinearEquationGroup}
\end{equation}
加入曲线上的全微分关系 $\dd u=u_x\dd x+u_y\dd y$、$\dd v=v_x\dd x+v_y\dd y$，得到
\[
 \underbrace{\begin{bmatrix}
 a_1&b_1&c_1&d_1\\a_2&b_2&c_2&d_2\\
 \dd x&\dd y&0&0\\0&0&\dd x&\dd y
 \end{bmatrix}}_{\mathcal M}
 \begin{bmatrix}u_x\\u_y\\v_x\\v_y\end{bmatrix}
 =\begin{bmatrix}f_1\\f_2\\\dd u\\\dd v\end{bmatrix}.
\]
令 $\det\mathcal M=0$ 得到特征方向。对 $m=\dd y/\dd x$，有
\[
 am^2+bm+c_0=0,\qquad
 \begin{cases}
 a=a_1c_2-a_2c_1,\\
 b=-(a_1d_2-a_2d_1+b_1c_2-b_2c_1),\\
 c_0=b_1d_2-b_2d_1.
 \end{cases}
\]
竖直方向用齐次式 $a(\dd y)^2+b\dd x\dd y+c_0(\dd x)^2=0$ 判断。
对不恒为零的特征二次式，令 $\Delta=b^2-4ac_0$：
\begin{center}
 \begin{tabular}{lll}
 \toprule
 判别式 & 每点的实特征方向 & 一阶二元系统的判断\\
 \midrule
 $\Delta>0$ & 两个不同方向 & 双曲型\\
 $\Delta=0$ & 一个重合方向 & 重特征，须进一步判断\\
 $\Delta<0$ & 无实方向 & 椭圆型\\
 \bottomrule
 \end{tabular}
\end{center}
\textbf{抛物型判据的适用范围：}二阶标量方程
$A_2\phi_{xx}+2B_2\phi_{xy}+C_2\phi_{yy}+\text{低阶项}=0$，
在主部非零时，$B_2^2-A_2C_2=0$ 对应抛物型；一阶系统不能仅凭重根判为抛物型。

\paragraph{3. 特征值法}
令 $\mathbf W=(u,v)^{\mathsf T}$，
$A=\begin{bmatrix}a_1&c_1\\a_2&c_2\end{bmatrix}$，
$B=\begin{bmatrix}b_1&d_1\\b_2&d_2\end{bmatrix}$，则
\[
 A\mathbf W_x+B\mathbf W_y=\mathbf f,\qquad
 C=A^{-1}B\quad(A\text{ 可逆}),\qquad \det(B-mA)=0.
\]
$C$ 的特征值就是特征斜率 $m$：\textbf{不同实根对应严格双曲型，非实共轭根对应椭圆型}。
重实根还要检查特征向量是否完整；\textbf{负实根只表示反向斜率，不代表椭圆型}。
多维系统需检查各实方向上的通量 Jacobian。

\paragraph{4. 三类 PDE 的核心性质}
\begin{itemize}
 \item \textbf{双曲型：}有限速度传播；按入射特征给定边界数据。
 \item \textbf{抛物型：}扩散与平滑；典型热方程沿时间推进，连续模型中传播速度无限。
 \item \textbf{椭圆型：}全域耦合的边值问题；局部变化通常影响整个区域。
\end{itemize}

\subsection{基础数值算法}
\label{sec:cfd-numerics}
\textbf{有限差分}近似微分算子；\textbf{有限体积}平衡控制体通量；\textbf{有限元}采用弱形式与基函数近似。
以下统一讨论均匀网格、光滑解，记 $u_i^n\approx u(x_i,t_n)$。

\subsubsection{有限差分方法基础}
由 Taylor 展开
\[
 u_{i\pm1}=u_i\pm\Delta x\,u_x+\frac{\Delta x^2}{2}u_{xx}
            \pm\frac{\Delta x^3}{6}u_{xxx}+\cdots
\]
得到常用公式：
\begin{align*}
 \text{向前：}\quad (u_x)_i&=\frac{u_{i+1}-u_i}{\Delta x}+O(\Delta x),\\
 \text{向后：}\quad (u_x)_i&=\frac{u_i-u_{i-1}}{\Delta x}+O(\Delta x),\\
 \text{中心：}\quad (u_x)_i&=\frac{u_{i+1}-u_{i-1}}{2\Delta x}+O(\Delta x^2),\\
 \text{二阶导数：}\quad (u_{xx})_i&=
       \frac{u_{i+1}-2u_i+u_{i-1}}{\Delta x^2}+O(\Delta x^2).
\end{align*}
二维混合导数为
\[
 (u_{xy})_{i,j}=\frac{u_{i+1,j+1}-u_{i+1,j-1}-u_{i-1,j+1}+u_{i-1,j-1}}
 {4\Delta x\Delta y}+O(\Delta x^2+\Delta y^2).
\]
\textbf{精度要点：}前／后向一阶，中心二阶；非均匀网格需重新求差分系数。

\subsubsection{显式方法与隐式方法}
\textbf{显式：}由已知时间层直接计算新时间层，单步简单，通常受较强时间步限制。
\textbf{隐式：}新时间层含待求未知量，需解代数方程组；可放宽某些稳定性限制，但每步代价较高。

对热方程 $T_t=\alpha T_{xx}$（$\alpha>0$），定义\textbf{扩散数}
$r=\alpha\Delta t/\Delta x^2$。

\paragraph{FTCS：时间向前、空间中心}
\begin{equation}
 T_i^{n+1}=T_i^n+r(T_{i+1}^n-2T_i^n+T_{i-1}^n).
 \label{eq:cfd-heat-ftcs}
\end{equation}
\textbf{显式；时间一阶、空间二阶；稳定条件 $0\le r\le1/2$。}

\paragraph{Crank--Nicolson：新旧时间层取平均}
记 $\delta_{xx}T_i=(T_{i+1}-2T_i+T_{i-1})/\Delta x^2$，则
\[
 \frac{T_i^{n+1}-T_i^n}{\Delta t}
  =\frac\alpha2(\delta_{xx}T_i^{n+1}+\delta_{xx}T_i^n).
\]
整理得
\begin{equation}
 -\frac r2T_{i-1}^{n+1}+(1+r)T_i^{n+1}-\frac r2T_{i+1}^{n+1}
 =\frac r2T_{i-1}^n+(1-r)T_i^n+\frac r2T_{i+1}^n.
 \label{eq:cfd-cn}
\end{equation}
\textbf{隐式；时间、空间均二阶；对上述热方程无条件稳定。}
给定两端温度后形成三对角系统，可用 Thomas 算法求解。
\textbf{稳定不等于准确或无振荡}：CN 时间步过大时仍可能出现交替振荡。

\subsubsection{误差与稳定性分析}
\label{sec:cfd-stability}
\paragraph{1. 误差传播与 von Neumann 分析}
\textbf{离散误差}来自用离散方程近似 PDE，\textbf{舍入误差}来自有限精度运算。
若线性格式为 $D^{n+1}=QD^n+b^n$，则扰动满足
\[
 \epsilon^{n+1}=Q\epsilon^n+\eta^n,
\]
其中 $\eta^n$ 为本步新误差；忽略 $\eta^n$ 后，误差满足齐次离散方程。
\textbf{稳定性要求扰动在固定时间区间内一致有界，不要求每一步都衰减。}

von Neumann 方法用于\textbf{线性、常系数、均匀网格的周期／无限域问题}：
\[
 \epsilon_i^n=\sum_m\widehat\epsilon_m^n e^{\mathrm i k_mx_i},\qquad
 \widehat\epsilon_m^{n+1}=g(\theta_m)\widehat\epsilon_m^n,
 \quad\theta_m=k_m\Delta x.
\]
\textbf{步骤：代入单个 Fourier 模态 $\to$ 求放大因子 $g$ $\to$ 检查所有波数。}
对下述无物理增长的标量单步格式，采用 $|g(\theta)|\le1$ 的不放大判据；
多步格式还要检查全部根及单位圆重根，边界与非线性效应需另行分析。
热方程的结果为
\[
 g_{\mathrm{FTCS}}=1-4r\sin^2(\theta/2),\qquad
 g_{\mathrm{CN}}=\frac{1-2r\sin^2(\theta/2)}{1+2r\sin^2(\theta/2)}.
\]
前者要求 $0\le r\le1/2$，后者对任意 $r\ge0$ 均满足 $|g|\le1$。

\paragraph{2. CFL 条件}
对 $u_t+cu_x=0$，定义带符号的\textbf{Courant 数}
\[
 \nu=\frac{c\Delta t}{\Delta x},\qquad
 |\nu|=\frac{\text{一个时间步的信息传播距离}}{\text{网格间距}}.
\]
\[
 \boxed{\text{数值依赖域必须包含连续问题的依赖域}}
\]
每步只跨一个网格的显式格式要求 $|\nu|\le1$。
\textbf{CFL 是必要条件，一般不是充分条件；具体界限取决于格式。}
Euler 系统还需计入声速，采用相应方向上的最大特征速度。

\paragraph{3. 常用对流差分格式}
以下均针对常速对流 $u_t+cu_x=0$，稳定性按通常固定 Courant 数的加密方式讨论。

\textbf{FTCS}
\[
 u_i^{n+1}=u_i^n-\frac\nu2(u_{i+1}^n-u_{i-1}^n),\qquad
 g=1-\mathrm i\nu\sin\theta.
\]
$|g|^2=1+\nu^2\sin^2\theta$：\textbf{纯对流下不稳定}，即使满足 CFL 也不能保证稳定。

\textbf{Leap-Frog（CTCS）}
\[
 u_i^{n+1}=u_i^{n-1}-\nu(u_{i+1}^n-u_{i-1}^n),\qquad
 g^2+2\mathrm i\nu\sin\theta\,g-1=0.
\]
\textbf{时间、空间均二阶；$|\nu|<1$ 时无幅值耗散，但存在色散和计算模态。}
需要两层数据及启动步骤；$|\nu|=1$ 的重根需单独处理，实际通常留有稳定裕量。
用于热方程时，Leap-Frog 不稳定。

\textbf{Lax（Lax--Friedrichs）}
\[
 u_i^{n+1}=\frac{u_{i+1}^n+u_{i-1}^n}{2}
          -\frac\nu2(u_{i+1}^n-u_{i-1}^n),\qquad
 g=\cos\theta-\mathrm i\nu\sin\theta.
\]
\textbf{固定非零 Courant 数下一阶；$|\nu|\le1$ 时稳定；耗散明显，会抹平陡峭梯度。}
一致性还要求 $\Delta x^2/\Delta t\to0$。

\textbf{Lax--Wendroff}
核心技巧是\textbf{用 PDE 将时间导数换成空间导数}：
\[
 u^{n+1}=u^n+\Delta t\,u_t+\tfrac12\Delta t^2u_{tt}+\cdots,
 \qquad u_t=-cu_x,\quad u_{tt}=c^2u_{xx}.
\]
再用空间中心差分，得到
\begin{equation}
 u_i^{n+1}=u_i^n-\frac\nu2(u_{i+1}^n-u_{i-1}^n)
          +\frac{\nu^2}{2}(u_{i+1}^n-2u_i^n+u_{i-1}^n).
 \label{eq:cfd-lax-wendroff}
\end{equation}
\[
 g=1-\mathrm i\nu\sin\theta+\nu^2(\cos\theta-1).
\]
\textbf{时间、空间均二阶；$|\nu|\le1$ 时稳定；间断附近可能出现色散振荡。}

\paragraph{4. 一致性、稳定性、收敛性}
\begin{itemize}
 \item \textbf{一致性：}离散算子的截断误差随网格细化趋于0。
 \item \textbf{稳定性：}扰动传播受到与网格无关的控制。
 \item \textbf{收敛性：}离散解随网格细化趋于连续问题的解。
\end{itemize}
Lax 等价定理适用于\textbf{适定线性初值问题的一致线性差分逼近}：
\[
 \boxed{\text{一致性}+\text{稳定性}\ \Longrightarrow\ \text{收敛性}}.
\]
在上述前提下稳定性与收敛性等价；不能直接推广为任意非线性 PDE 的定理。

\paragraph{5. 修正方程、耗散、色散与人工黏性}
\textbf{修正方程}将离散格式写成“原 PDE $+$ 网格相关高阶项”，用于识别主导误差：
\begin{align*}
 \text{FTCS:}\quad u_t+cu_x
   &=-\tfrac12c^2\Delta t\,u_{xx}+\cdots
      &&\text{负扩散},\\
 \text{Lax:}\quad u_t+cu_x
   &=\frac{\Delta x^2}{2\Delta t}(1-\nu^2)u_{xx}+\cdots
      &&\text{稳定区间内正扩散},\\
 \text{Lax--Wendroff:}\quad u_t+cu_x
   &=\frac{c\Delta x^2}{6}(\nu^2-1)u_{xxx}+\cdots
      &&\text{主导色散}.
\end{align*}
\begin{itemize}
 \item \textbf{耗散（dissipation）：}改变振幅，衰减短波并抹平梯度。
 \item \textbf{色散（dispersion）：}改变不同波数的传播速度，造成相位误差与振荡。
\end{itemize}
对实常系数，偶数阶导数控制振幅，奇数阶导数控制相位；耗散项还必须具有正确符号：
\[
 u_t+cu_x=\epsilon_2u_{xx}-\epsilon_4u_{xxxx},\qquad
 \epsilon_2,\epsilon_4\ge0.
\]
其 Fourier 振幅衰减率为 $\epsilon_2k^2+\epsilon_4k^4$，故\textbf{二阶项取正号、四阶项取负号}；
四阶耗散更集中地抑制短波。
\textbf{人工黏性}是数值引入的耗散，不是流体的物理黏性；应兼顾稳定性、精度与离散守恒，
并在趋向原无黏方程的加密极限中适当消失。

\subsubsection{计算网格}
\label{sec:cfd-grid}
\textbf{结构网格}有规则逻辑索引，\textbf{非结构网格}以单元连接关系组织。
网格应贴合边界、在高梯度处加密、尺寸平缓变化，并避免严重扭曲或折叠。

\paragraph{1. 物理平面与计算平面}
\textbf{固定、光滑、可逆的坐标映射}将物理平面中的曲线非均匀网格转换为计算平面的规则网格：
\[
 x=x(\xi,\eta),\qquad y=y(\xi,\eta).
\]
定义
\begin{equation}
 D=x_\xi y_\eta-x_\eta y_\xi,\qquad
 J=\det\frac{\partial(\xi,\eta)}{\partial(x,y)}=\frac1D.
 \label{eq:cfd-jacobian}
\end{equation}
\textbf{本讲采用课件的 $J$；Anderson 的 Jacobian 对应这里的 $D$。}
局部可逆要求 $D\ne0$，保向网格取 $D>0$；面积元为 $\dd x\dd y=J^{-1}\dd\xi\dd\eta$。
度量与导数变换为
\[
 \begin{bmatrix}\xi_x&\xi_y\\\eta_x&\eta_y\end{bmatrix}
 =J\begin{bmatrix}y_\eta&-x_\eta\\-y_\xi&x_\xi\end{bmatrix},
\]
\[
 \partial_x=J(y_\eta\partial_\xi-y_\xi\partial_\eta),\qquad
 \partial_y=J(-x_\eta\partial_\xi+x_\xi\partial_\eta).
\]

\paragraph{2. 曲线坐标下的二维 Euler 方程}
对无体源的二维无黏流动，
\[
 \mathbf U_t+\mathbf F_x+\mathbf G_y=0,
 \quad
 \mathbf U=\begin{bmatrix}\rho\\\rho u\\\rho v\\E_t\end{bmatrix},\quad
 \mathbf F=\begin{bmatrix}\rho u\\\rho u^2+p\\\rho uv\\u(E_t+p)\end{bmatrix},\quad
 \mathbf G=\begin{bmatrix}\rho v\\\rho uv\\\rho v^2+p\\v(E_t+p)\end{bmatrix}.
\]
这里 $E_t=\rho[e+(u^2+v^2)/2]$。固定网格下，变换后仍保持强守恒形式：
\begin{equation}
 \boxed{\widetilde{\mathbf U}_t+
        \widetilde{\mathbf F}_\xi+\widetilde{\mathbf G}_\eta=0},
 \label{eq:cfd-curvilinear-euler}
\end{equation}
\begin{align*}
 \widetilde{\mathbf U}&=\frac{\mathbf U}{J},\\
 \widetilde{\mathbf F}&=\frac{\xi_x\mathbf F+\xi_y\mathbf G}{J}
                  =y_\eta\mathbf F-x_\eta\mathbf G,\\
 \widetilde{\mathbf G}&=\frac{\eta_x\mathbf F+\eta_y\mathbf G}{J}
                  =-y_\xi\mathbf F+x_\xi\mathbf G.
\end{align*}
其关键是混合偏导交换使度量项相消：
$\partial_\xi y_\eta-\partial_\eta y_\xi=0$、
$-\partial_\xi x_\eta+\partial_\eta x_\xi=0$。
离散时应兼容这些恒等式，以保持均匀来流。
